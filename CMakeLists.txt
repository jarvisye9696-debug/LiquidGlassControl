cmake_minimum_required(VERSION 3.20)
project(LiquidGlassControl LANGUAGES CXX RC)
add_executable(LiquidGlassControl WIN32 main.cpp assets/app.rc)
target_include_directories(LiquidGlassControl PRIVATE assets)
target_compile_features(LiquidGlassControl PRIVATE cxx_std_17)
target_link_libraries(LiquidGlassControl PRIVATE dwmapi shell32 user32 gdi32 advapi32 ole32 comctl32 uxtheme)
if(MSVC)
  set_property(TARGET LiquidGlassControl PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(LiquidGlassControl PRIVATE /utf-8 /W4)
else()
  target_link_options(LiquidGlassControl PRIVATE -municode -static)
  target_compile_options(LiquidGlassControl PRIVATE -Wall -Wextra)
endif()
