cmake_minimum_required(VERSION 3.20)
project(LiquidGlassControl LANGUAGES CXX RC)
add_library(GlassEngine SHARED glass_engine.cpp glass_engine.def)
target_compile_features(GlassEngine PRIVATE cxx_std_20)
set(LIQUIDGLASS_WINRT_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/build/winrt/include" CACHE PATH "Generated C++/WinRT and WinUI headers (prepare-winrt.ps1)")
if(EXISTS "${LIQUIDGLASS_WINRT_INCLUDE}/winrt/Microsoft.UI.Xaml.Media.h")
  target_include_directories(GlassEngine BEFORE PRIVATE "${LIQUIDGLASS_WINRT_INCLUDE}")
elseif(MSVC)
  message(FATAL_ERROR "Run pwsh ./prepare-winrt.ps1 before configuring.")
endif()
target_link_libraries(GlassEngine PRIVATE ole32 oleaut32 uuid runtimeobject comctl32 dwmapi advapi32 d2d1)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/engine-$<CONFIG>.rc" CONTENT "201 RCDATA \"$<TARGET_FILE:GlassEngine>\"\n202 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/release/Third Party Notices.txt\"\n")
add_executable(LiquidGlassControl WIN32 main.cpp assets/app.rc)
target_sources(LiquidGlassControl PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/engine-$<CONFIG>.rc")
add_dependencies(LiquidGlassControl GlassEngine)
target_include_directories(LiquidGlassControl PRIVATE assets)
target_compile_features(LiquidGlassControl PRIVATE cxx_std_17)
target_link_libraries(LiquidGlassControl PRIVATE dwmapi shell32 user32 gdi32 advapi32 ole32 comctl32 uxtheme uuid)
if(MSVC)
  set_property(TARGET LiquidGlassControl PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_property(TARGET GlassEngine PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(GlassEngine PRIVATE /utf-8 /W4)
  target_compile_options(LiquidGlassControl PRIVATE /utf-8 /W4)
else()
  target_link_options(LiquidGlassControl PRIVATE -municode -static)
  target_compile_options(LiquidGlassControl PRIVATE -Wall -Wextra)
  target_link_options(GlassEngine PRIVATE -static)
  target_compile_options(GlassEngine PRIVATE -Wall -Wextra)
endif()
